Searchables/

│       ├── shell/              # Sign-in page, activity bar, tab bar, status bar
│       ├── documents/          # Search, PDF page view, inspector, contents, glossary, compare, pins, library
│       └── code/               # Code panel (search, files, dependencies, APIs), editor, inspector

│   │   ├── uninstall-service.sh    # Removes the service; keeps the library
│   │   ├── common.sh               # Shared helpers: Python 3.10+ lookup, .venv, Tesseract
│   │   └── README.md               # macOS setup, security and troubleshooting
│   └── windows/
│       ├── run.ps1                 # Run in a PowerShell window (sets up .venv on first use)
│       ├── install-service.ps1     # Installs/updates the Windows service (WinSW), venv, data folder
│       ├── uninstall-service.ps1   # Removes the service; keeps the library unless -RemoveData
│       └── README.md               # Windows setup, service, security and troubleshooting
├── src/
│   ├── __init__.py
│   ├── config.py               # Constants, thresholds, and paths
│   ├── auth.py                 # The password (scrypt hash) and signed-in browser sessions
│   ├── server/
│   │   ├── __main__.py         # python -m src.server: starts the app
│   │   ├── app.py              # The JSON API over the library, collections, code index and indexer; serves the front end

│   ├── code/                   # Source code browsing (the Code view)
│   │   ├── __init__.py
│   │   ├── parser.py           # One Java / JS / TS file → symbols, references, imports, HTTP endpoints (tree-sitter)
│   │   ├── store.py            # The code index: scanning, linking references to declarations, usages, call trees, dependencies, endpoints
│   │   ├── manifests.py        # Declared libraries (package.json, pom.xml, build.gradle) and tsconfig import aliases
│   │   └── models.py           # Codebase, Symbol, Usage, Dependency, Endpoint, …
│   ├── extractor/
│   │   ├── __init__.py
│   │   ├── pdf.py              # PyMuPDF ingestion, OCR & block extraction
│   │   ├── identifiers.py      # Identifier discovery: candidates, families by shape, roles
│   │   ├── objects.py          # Tables, figures (via captions) & cross-reference detection
│   │   ├── provisions.py       # shall / should / may / NOTE classification; list items under a lead-in
│   │   └── structure.py        # Clause headings & running header/footer detection
│   ├── search/
│   │   ├── __init__.py
│   │   ├── indexer.py          # Background indexing queue (add, re-index, add meaning search), resumed after a restart
│   │   ├── library.py          # Persistent library: documents, blocks, tables, xrefs, history, keyword + meaning search, related passages

│   │   └── query.py            # User query → safe FTS5 query (phrases, OR, NOT, prefix; all or some of the words)
│   ├── research/
│   │   ├── __init__.py
│   │   ├── assess.py           # Same-topic assessment rules: values, shall/should/may, negation, identifiers

│   │   ├── compare.py          # Edition comparison: clause pairing, word diffs, provision changes
│   │   ├── export.py           # Collection → Markdown / Word with citations
│   │   └── glossary.py         # Defined terms from "Terms and definitions" clauses
│   ├── ui/                     # The earlier Streamlit interface's screens (see app.py)

    └── codebase/               # A small Java + TypeScript/JavaScript project the code tests index
